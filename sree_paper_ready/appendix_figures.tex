\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/accum_probe_f1.png}}
\caption{Accumulator variable probe accuracy and macro-F1 vs.\ layer for Python
XLCoST programs, Qwen2.5-1.5B, across six variable-naming strategies. The
baseline peaks at F1~$= 0.893$ (L9); all-same and misleading-name strategies
exceed the baseline, confirming that structural context dominates lexical cues
for this role.}
\label{fig:appendix-sree-accum-probe}
\end{figure}

\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/accum_delta_f1.png}}
\caption{$\Delta$ macro-F1 relative to baseline naming for each perturbation
strategy, accumulator role. Positive bars (all-same~$+0.076$,
numeric~$+0.093$, misleading~$+0.079$) indicate that removing or scrambling
variable names \emph{improves} accumulator detection, in sharp contrast to the
index role where misleading names cause $\Delta\text{F1} = {-}0.272$.}
\label{fig:appendix-sree-accum-delta}
\end{figure}

\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/accum_cross_language.png}}
\caption{Cross-language transfer for the accumulator probe. A Python-trained
probe (L9) achieves $0.943$--$0.948$ accuracy on Java, C++, C, and C\#,
indicating a language-agnostic accumulator direction in the residual stream.
In-domain F1 for compiled languages ($0.939$--$0.970$) exceeds Python ($0.893$)
due to the syntactic regularity of \texttt{+=} in C-family loop bodies.}
\label{fig:appendix-sree-accum-crosslang}
\end{figure}

\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/index_probe_f1.png}}
\caption{Index/key variable probe accuracy and macro-F1 vs.\ layer, Python
XLCoST, Qwen2.5-1.5B, baseline (original variable names). Peak F1~$= 0.959$
at L8 exceeds the accumulator role, reflecting the strong lexical regularity
of index-style names (\texttt{i}, \texttt{j}, \texttt{idx}).}
\label{fig:appendix-sree-index-probe}
\end{figure}

\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/index_renamed_comparison.png}}
\caption{Index/key probe F1 under original versus misleading variable names.
Misleading renaming causes $\Delta\text{F1} = {-}0.272$, the largest drop of
any role studied, indicating that index role detection depends heavily on
surface lexical identity.}
\label{fig:appendix-sree-index-renamed}
\end{figure}

\begin{figure}[ht]
\centering
\includegraphics[width=0.9\linewidth]{\detokenize{sree's experiments/index_cross_language.png}}
\caption{Cross-language transfer for the index/key probe. A Python-trained probe
achieves $0.973$--$0.988$ accuracy on Java and C++, reflecting the syntactic
regularity of loop-index patterns across languages and exceeding the accumulator
transfer range ($0.943$--$0.948$).}
\label{fig:appendix-sree-index-crosslang}
\end{figure}
